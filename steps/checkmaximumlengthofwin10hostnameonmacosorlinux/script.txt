case "{operatingSystemName}" in
    *Server*)
        echo "Server hostname length is not limited to 15, skipping."
        ;;
    *)
        HOSTNAME="{newMachine.hostname}"
        LENGTH="${#HOSTNAME}"
        MAX_LENGTH="15"
        if [ "${LENGTH}" -gt "${MAX_LENGTH}" ]
        then
            echo "Length of hostname ${LENGTH} for Windows workstation can't be more than 15."
            false
        else
            echo "Length of hostname ${LENGTH} is less than or equal to 15."
        fi
        ;;
esac
