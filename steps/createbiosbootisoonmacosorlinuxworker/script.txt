cd "{linuxWorkerDirectory}build-{newMachine.fqn}"

if [ "{biosOrUefi}" = "BIOS" ]
then
    if uname | \
        grep -iq "darwin"
    then
        echo "is macOS"
        export PATH="${HOME}/Parallels/homebrew/bin:$PATH"
        echo "Using $(which xorriso)"

        xorriso -as mkisofs \
            --norock \
            -graft-points \
            -b boot.img \
            -no-emul-boot \
            -boot-load-size 8 \
            -c BOOT.CAT \
            -iso-level 3 \
            -J \
            -joliet-long \
            -relaxed-filenames \
            -full-iso9660-filenames \
            -disable-deep-relocation \
            -omit-version-number \
            -omit-period \
            -m bootmgr \
            -V "KS_WIN" \
            -o "../unattended_{newMachine.fqn}.iso" \
            BOOTMGR=./bootmgr \
            .

        xorriso -indev "../unattended_{newMachine.fqn}.iso" -report_el_torito plain

        xorriso -indev "../unattended_{newMachine.fqn}.iso" -ls /

        xorriso -read_fs ecma119 -indev "../unattended_win03.localdomain.iso" -ls /

    else
        echo "not macOS"

        mkisofs \
            -b boot.img \
            -no-emul-boot \
            -c BOOT.CAT \
            -iso-level 3 \
            -udf \
            -joliet-long \
            -relaxed-filenames \
            -allow-limited-size \
            -J \
            -full-iso9660-filenames \
            -disable-deep-relocation \
            -omit-version-number \
            -V "KS_WIN" \
            -o "../unattended_{newMachine.fqn}.iso" \
            .
    fi

else
    echo "Skipping for UEFI boot."
fi
