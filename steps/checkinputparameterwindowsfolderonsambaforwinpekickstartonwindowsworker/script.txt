if ("{isWinPE}" -ine "true") {
    Write-Host "Please set isWinPE to the string 'true'"
    exit 1
}

if ("{sambaServerDirectory}" -eq "windows10") {
    if ("{operatingSystemName}" -match "Server") {
        Write-Host "Samba folder windows10 requires Operating System Name Windows 10 or Windows 11."
        exit 1
    }
}

if (("{sambaServerDirectory}" -eq "windows2016") -or
    ("{sambaServerDirectory}" -eq "windows2019") -or
    ("{sambaServerDirectory}" -eq "windows2022")) {
    if ("{operatingSystemName}" -notmatch "Server") {
        Write-Host "Samba Server folder requires Operating System Name to contain Server."
        exit 1
    }
}

$uncPath = "\\{sambaServerIpAddress}\share\{sambaServerDirectory}"

if (Test-Path $uncPath) {
    Write-Host "The SMB share is accessible. Listing contents:"
    Get-ChildItem $uncPath
} else {
    Write-Host "The SMB share is not accessible or does not exist."
}
