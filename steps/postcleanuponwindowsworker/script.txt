$DIR="{windowsWorkerDirectory}"

if (Test-Path $DIR) {
    Set-Location "${DIR}"
} else {
    New-Item -Path '${DIR}' -ItemType Directory
    Test-Path '${DIR}'
}

$DIR="build-{newMachine.fqn}"

if (Test-Path $DIR) {
    Remove-Item -Recurse $DIR -Force
} else {
    Write-Host "No build directory to remove."
}

# The wildcard will allow removal of 
# - {isoFilename}
# - drivers_{newMachine.fqn}.iso
$ISOS="*{newMachine.fqn}.iso"


if (test-Path "${ISOS}" -pathtype leaf) {
	Get-ChildItem "${ISOS}" | `
    	Foreach-Object {
    		if (test-Path $_.Name -pathtype leaf) {
    			Write-Host $_.Name
    			Remove-Item $_.Name
    		}
    	}
} else {
	Write-Host "No ISOs to remove, skipping."
}
