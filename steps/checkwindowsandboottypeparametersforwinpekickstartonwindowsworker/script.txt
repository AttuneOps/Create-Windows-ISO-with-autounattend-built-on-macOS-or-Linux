Write-Host "biosOrUefi = {biosOrUefi}"
Write-Host "operatingSystemName = {operatingSystemName}"

if ("{biosOrUefi}" -ine "BIOS" -and "{biosOrUefi}" -ine "UEFI") {
    Write-Host "FAIL: BIOS or UEFI must be BIOS or UEFI"
    exit 1
}

if ([string]::IsNullOrWhiteSpace("{operatingSystemName}")) {
    Write-Host "FAIL: Operating System Name must be set"
    exit 1
}

Write-Host "SUCCESS: The parameters above are set correctly"
