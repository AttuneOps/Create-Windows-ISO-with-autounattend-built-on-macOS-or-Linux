cd {linuxWorkerDirectory}build-{newMachine.fqn}

if ! command -v 7z >/dev/null 2>&1
then
    echo "Error: 7z is not installed." >&2
    exit 1
fi

# Case insensitve globbing
shopt -s nocaseglob

if ls | grep -q *iso
then
    
    ISO="$(echo *iso)"
    echo "ISO is $ISO"
    
    7z x "${ISO}"
    
    rm -v "${ISO}"
else
    echo "ISO is already extracted."
fi
