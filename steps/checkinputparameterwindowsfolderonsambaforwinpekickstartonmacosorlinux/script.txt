if [ "{isWinPE}" != "true" ] && [ "{isWinPE}" != "TRUE" ]
then
    echo "Please set isWinPE to the string 'true'"
    exit 1
fi

if [ "{sambaServerDirectory}" = "windows10" ]
then
    case "{operatingSystemName}" in
        *Server*)
            echo "Samba folder windows10 requires Operating System Name Windows 10 or Windows 11."
            exit 1
            ;;
    esac
fi

if [ "{sambaServerDirectory}" = "windows2016" ] ||
    [ "{sambaServerDirectory}" = "windows2019" ] ||
    [ "{sambaServerDirectory}" = "windows2022" ]
then
    case "{operatingSystemName}" in
        *Server*)
            ;;
        *)
            echo "Samba Server folder requires Operating System Name to contain Server."
            exit 1
            ;;
    esac
fi
