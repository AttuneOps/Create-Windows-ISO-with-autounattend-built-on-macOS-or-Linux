BOOT="$(echo "{biosOrUefi}" | tr "[:lower:]" "[:upper:]")"
echo "biosOrUefi = {biosOrUefi}"
echo "operatingSystemName = {operatingSystemName}"

if [ "$BOOT" != "BIOS" ] && [ "$BOOT" != "UEFI" ]
then
    echo "FAIL: BIOS or UEFI must be BIOS or UEFI" >&2
    false
fi

if [ -z "{operatingSystemName}" ]
then
    echo "FAIL: Operating System Name must be set" >&2
    false
fi

echo "SUCCESS: The parameters above are set correctly"
